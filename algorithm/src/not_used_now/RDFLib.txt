        clase_name = item["type"].capitalize()
        g.add((sujeto, RDF.type, RICO[clase_name]))

        if "identifier" in item:

        if "creator_id" in item:

    return g

def validar_con_shacl(data_graph, ontologia_path):
    # 2. CARGAR LA ONTOLOGÍA LOCAL
    rico_ont_graph = Graph()
    if os.path.exists(ontologia_path):
        print(f"--- Cargando ontología local: {ontologia_path} ---")
        # El formato puede ser 'xml' para .rdf o 'turtle' para .ttl
        formato = "xml" if ontologia_path.endswith(".rdf") else "turtle"
        rico_ont_graph.parse(ontologia_path, format=formato)
    else:
        print(f"⚠️ No se encontró {ontologia_path}. La validación será sin ontología.")
        rico_ont_graph = None

    # Definimos reglas SHACL
    shapes_graph = f"""
    @prefix sh: <http://www.w3.org/ns/shacl#> .
    @prefix rico: <{RICO}> .
    @prefix ex: <{BASE}> .

    ex:RecordShape
        a sh:NodeShape ;
        sh:targetClass rico:Record ;
        sh:property [

            sh:minCount 1 ;

Esta es una arquitectura muy interesante. Al introducir un **LLM (GPT-OSS)** en el bucle, el riesgo de "alucinaciones" semánticas aumenta, lo que hace que el enfoque que hemos discutido (RDFLib + SHACL) sea **aún más necesario** que antes.

Tu método actual es vulnerable porque confías en que el LLM devuelva claves de JSON que coincidan exactamente con la ontología RiC-O. Si el LLM escribe `rico:CreationDate` en lugar de `rico:isAssociatedWithDate`, tu clase `RDFconverter` actual lo dará por bueno, pero tu grafo será semánticamente inválido.

Aquí tienes el análisis y los ajustes sugeridos para tu flujo:

---

### 1. El Riesgo: El "Gap" entre el LLM y RiC-O

Los LLM tienden a simplificar. Aunque le pidas RiC-O estricto, a veces inventará atributos.

* **Enfoque actual:** Tu `RDFconverter` simplemente traduce lo que el LLM escupe. Es un "mapeo ciego".
* **Enfoque sugerido:** Usar **RDFLib** como un "filtro de integridad" después del LLM.

### 2. Mejoras Sugeridas en el código principal

#### A. Integrar la Validación en el `run()`

No esperes al final para saber si el LLM falló. Valida el resultado del LLM antes de retornar el resultado del algoritmo.

```python
# ... dentro de tu función run() ...

```

#### B. Refinar el Prompt para RDFLib

Para que `crear_grafo_rico` funcione bien con RDFLib, el LLM debe ser muy consistente con los `@id`. Tu prompt ya pide `@id: 001`, lo cual es perfecto para que RDFLib genere URIs consistentes.

### 3. ¿Por qué cambiar el enfoque de `RDFconverter`?

Tu clase `RDFconverter` actual hace limpieza de strings (`rdf_key_name`, `rdf_entity_name`). Esto es **peligroso** con LLMs porque:

1. Si el LLM devuelve `rico:Has Creator` (con espacio), tu función `rdf_key_name` lo convertirá a `hasCreator`.
2. Pero si el LLM devuelve algo totalmente erróneo como `rico:quienLoHizo`, tu función lo convertirá a `quienLoHizo`, y terminarás con un RDF que **no es RiC-O**.

**El enfoque profesional con LLM es:**

1. **Extraer** (LLM).
2. **Validar y Mapear** (RDFLib + un diccionario de mapeo estricto).
3. **Rechazar/Corregir** (SHACL).

---

### 4. Ajuste en tu `save()`

Como estás usando `ocean-runner`, es vital que los errores de validación queden registrados. Sugiero guardar un archivo adicional `validation_log.txt`:

### Conclusión sobre tu enfoque:

Tu arquitectura es sólida para un sistema de extracción automática de archivos. Sin embargo, para que sea **"Fiable" (como pides)**, debes sustituir la clase `RDFconverter` (que es una manipulación de texto) por un procesador basado en **RDFLib** que cargue la ontología real.

    def _cargar_ontologia(self):
        """Carga la ontología RiC-O local para habilitar inferencia y sugerencias."""
        g = Graph()
        if os.path.exists(self.ontologia_path):
            # Detectar formato por extensión
            formato = "xml" if self.ontologia_path.endswith(".rdf") else "turtle"
            try:
                g.parse(self.ontologia_path, format=formato)
                return g
            except Exception as e:
                print(f"Error al parsear la ontología: {e}")
        return None

    def _extraer_propiedades_ontologia(self):
        """Lista todas las propiedades de RiC-O para comparar con lo generado por el LLM."""
        if not self.ont_graph: return []
        # Extraer nombres locales de los predicados que pertenecen al namespace RICO
        return [str(p).split("#")[-1] for p in self.ont_graph.predicates() if str(p).startswith(str(self.RICO))]

    def json_to_rdflib_graph(self, json_input):
        """
        Convierte el JSON del LLM en un Grafo RDFLib.
        Mantiene fielmente los nombres de atributos que el LLM ha generado.
        """
        g = Graph()
        g.bind("rico", self.RICO)
        g.bind("ex", self.BASE)

        # Cargar JSON (maneja string del LLM o diccionario)
        data = json.loads(json_input) if isinstance(json_input, str) else json_input

        for clase_tipo, entidades in data.items():
            for entidad in entidades:
                # 1. Crear Sujeto (URI basada en @id)
                sujeto = self.BASE[entidad["@id"]]
                
                # 2. Asignar Clase (@type)
                # Nota: El prompt pide UpperCamelCase, RDFLib construye la URI
                tipo_uri = self.RICO[entidad["@type"]]
                g.add((sujeto, RDF.type, tipo_uri))

                # 3. Procesar Atributos y Relaciones
                for llave, valor in entidad.items():
                    if llave in ["@id", "@type"]:
                        continue
                    
                    # Limpiar el prefijo si el LLM lo incluyó en el nombre de la llave
                    nombre_prop = llave.replace("rico:", "")
                    predicado = self.RICO[nombre_prop]

                    # Tratamiento de valores
                    if isinstance(valor, str) and valor.isdigit() and len(valor) == 3:
                        # Si es un ID de 3 dígitos, asumimos que es una relación (URI)
                        g.add((sujeto, predicado, self.BASE[valor]))
                    elif isinstance(valor, list):
                        for item in valor:
                            obj = self.BASE[item] if (isinstance(item, str) and item.isdigit()) else Literal(item)
                            g.add((sujeto, predicado, obj))
                    else:
                        # Por defecto, tratar como Literal (texto/dato)
                        g.add((sujeto, predicado, Literal(valor)))
        return g

        """
        Valida el grafo contra SHACL e informa de propiedades que no existen en RiC-O.

        # Reglas básicas de integridad para registros archivísticos
        shapes_graph = f"""

        @prefix rico: <{self.RICO}> .
        @prefix ex: <{BASE}> .

            sh:targetClass rico:Record ;
            sh:property [
                sh:path rico:identifier ;
                sh:minCount 1 ;
                sh:message "Falta el identificador obligatorio en el Record." ;
            ] .
        """

        conforms, _, results_text = validate(
            data_graph,

            ont_graph=self.ont_graph,
            inference='rdfs', # Usa la jerarquía de la ontología cargada
            abort_on_first=False
        )

        # Generar sugerencias si se encuentran errores de nomenclatura
        sugerencias = []
        if not conforms:
            for p in data_graph.predicates():
                nombre_p = str(p).split("#")[-1]
                # Si es una propiedad de RICO pero no está en la ontología oficial
                if str(p).startswith(str(self.RICO)) and nombre_p not in self.propiedades_validas:
                    coincidencia = difflib.get_close_matches(nombre_p, self.propiedades_validas, n=1, cutoff=0.5)
                    if coincidencia:
                        sugerencias.append(f"❓ Propiedad 'rico:{nombre_p}' no encontrada. ¿Quizás es 'rico:{coincidencia[0]}'?")

        return conforms, results_text, sugerencias

                        g.add((subject, predicate, self.BASE[value]))
                    elif isinstance(value, list):
                        for item in value:
                            obj = self.BASE[item] if (isinstance(item, str) and item.isdigit()) else Literal(item)
                            g.add((subject, predicate, obj))
                    else:
                        # Default to Literal (text/data)
                        g.add((subject, predicate, Literal(value)))
        return g

    def validate_and_report(self, data_graph):
        """
        Validates the graph against SHACL and flags properties 
        that do not exist in the official RiC-O ontology.
        """
        # Basic integrity rules for archival records
        shapes_graph = f"""
        @prefix sh: <http://www.w3.org/ns/shacl#> .
        @prefix rico: <{self.RICO}> .
        @prefix ex: <{self.BASE}> .

        ex:RecordResourceShape
            a sh:NodeShape ;
            sh:targetClass rico:Record ;
            sh:property [
                sh:path rico:identifier ;
                sh:minCount 1 ;
                sh:message "Missing mandatory identifier in Record." ;
            ] .
        """

        conforms, _, results_text = validate(
            data_graph,
            shacl_graph=shapes_graph,
            ont_graph=self.ont_graph,
            inference='rdfs', # Uses the ontology hierarchy
            abort_on_first=False
        )

        # Generate suggestions for nomenclature errors
        suggestions = []
        ontology_errors_found = False
        
        for p in data_graph.predicates():
            name = str(p).split("#")[-1]
            if str(p).startswith(str(self.RICO)) and name not in self.valid_properties:
                ontology_errors_found = True
                match = difflib.get_close_matches(name, self.valid_properties, n=1, cutoff=0.5)
                if match:
                    suggestions.append(f"Property 'rico:{name}' invalid. Try 'rico:{match[0]}'")
                else:
                    suggestions.append(f"Property 'rico:{name}' not in RiC-O 1.1.")
        
        
        # We return the output even if it's not perfect
        is_valid = conforms and not ontology_errors_found
        
        final_report = "--- SHACL REPORT ---\n" + results_text
        if suggestions:
            final_report += "\n\n--- ONTOLOGY SUGGESTIONS ---\n" + "\n".join(suggestions)
        
        return is_valid, final_report, suggestions
    






    def visualize_graph(self, data_graph, output_path="rdf_visualization.png"):
        """
        Generates a visual diagram of the RDF graph.
        Entities are nodes and properties are labeled edges.
        """
        import networkx as nx
        import matplotlib.pyplot as plt

        # Create a directed graph
        nx_graph = nx.MultiDiGraph()

        for s, p, o in data_graph:
            # Simplify labels for readability
            # s_label: end of the URI (e.g., '001')
            # p_label: the property name (e.g., 'hasCreator')
            # o_label: the object URI or the literal string
            s_label = str(s).split('/')[-1] if '/' in str(s) else str(s)
            p_label = str(p).split('#')[-1] if '#' in str(p) else str(p)
            
            if isinstance(o, URIRef):
                o_label = str(o).split('/')[-1] if '/' in str(o) else str(o)
            else:
                # For literals, we truncate to avoid huge nodes
                o_label = (str(o)[:20] + '...') if len(str(o)) > 20 else str(o)

            nx_graph.add_edge(s_label, o_label, label=p_label)

        # Draw the graph
        plt.figure(figsize=(12, 10))
        # 'spring_layout' helps to separate nodes automatically
        pos = nx.spring_layout(nx_graph, k=1.5, iterations=50)
        
        # Draw nodes and labels
        nx.draw(nx_graph, pos, with_labels=True, node_color='skyblue', 
                node_size=3000, font_size=8, font_weight='bold', 
                arrows=True, connectionstyle='arc3, rad = 0.1')
        
        # Draw edge labels (properties)
        edge_labels = {(u, v): d['label'] for u, v, d in nx_graph.edges(data=True)}
        nx.draw_networkx_edge_labels(nx_graph, pos, edge_labels=edge_labels, font_size=7)

        plt.title("RiC-O RDF Graph Extraction", fontsize=15)
        plt.savefig(output_path, bbox_inches='tight')
        plt.close()
        print(f"✅ Graph visualization saved to: {output_path}")
